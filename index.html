คัดลอกคำสั่ง SQL ด้านล่างไปวางใน Supabase SQL Editor เพื่อสร้างตาราง schedules, bookings, reports และ reviews ได้ทันที

          
-- สร้างตาราง Schedules, Bookings, Reports, Reviews
-- ดูไฟล์ตัวเต็มได้ที่ supabase_schema.sql ใน Repository
CREATE TABLE IF NOT EXISTS public.schedules (id VARCHAR PRIMARY KEY, day VARCHAR, time VARCHAR, origin VARCHAR, dest VARCHAR, price NUMERIC, ...);
CREATE TABLE IF NOT EXISTS public.bookings (ref_code VARCHAR PRIMARY KEY, queue_no VARCHAR, passenger_name VARCHAR, phone VARCHAR, seats JSONB, amount NUMERIC, ...);
CREATE TABLE IF NOT EXISTS public.reports (ticket_id VARCHAR PRIMARY KEY, type VARCHAR, details TEXT, phone VARCHAR, status VARCHAR, ...);
